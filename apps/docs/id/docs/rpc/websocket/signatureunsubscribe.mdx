<RequestExample>
  ```json WebSocket
  {
    "jsonrpc": "2.0",
    "id": 1,
    "method": "signatureUnsubscribe",

      0
    ]
  }
  ```
</RequestExample>

<ResponseExample>
  ```json Response

  }
  ```

</ParamField>

## Result

<ResponseField name="result" type="boolean">

</ResponseField>
