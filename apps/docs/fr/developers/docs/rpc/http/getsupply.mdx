Renvoie des informations sur l'approvisionnement actuel.

<RequestExample>
  ```bash cURL
  curl https://api.devnet.solana.com -s -X POST \
    -H "Content-Type: application/json" \
    -d '
    {
      "jsonrpc": "2.0",
      "id": 1,
      "method": "getSupply",
      "params": [
        {
          "commitment": "finalized"
        }
      ]
    }
  '
  ```

  ```ts Kit
  import { createSolanaRpc } from "@solana/kit";

  const rpc_url = "https://api.devnet.solana.com";
  const rpc = createSolanaRpc(rpc_url);

  let supply = await rpc.getSupply().send();

  console.log(supply);
  ```

  ```ts web3.js
  import { Connection, clusterApiUrl } from "@solana/web3.js";

  const connection = new Connection(clusterApiUrl("devnet"), "confirmed");

  let supply = await connection.getSupply();

  console.log(supply);
  ```
</RequestExample>

## Parameters

<ParamField body="0" type="object">
  Objet de configuration contenant les champs suivants :

  <Expandable title="properties" defaultOpen>
    <ParamField body="commitment" type="string">
      Le commitment décrit le niveau de finalisation d'un bloc à un moment donné. Voir
      [Configuration de l'engagement d'état](/developers/docs/rpc#configuring-state-commitment).
    </ParamField>

    <ParamField body="excludeNonCirculatingAccountsList" type="bool">
      Exclure la liste des comptes non circulants de la réponse
    </ParamField>
  </Expandable>
</ParamField>

## Result

<ResponseField name="result" type="object">
  Le résultat sera un objet JSON contenant :

  <Expandable title="properties" defaultOpen>
    <ResponseField name="circulating" type="u64">
      Approvisionnement en circulation en lamport
    </ResponseField>

    <ResponseField name="nonCirculating" type="u64">
      Offre non circulante en lamports
    </ResponseField>

    <ResponseField name="nonCirculatingAccounts" type="array">
      Un tableau d'adresses de comptes non circulants, sous forme de chaînes de
      caractères. Si `excludeNonCirculatingAccountsList` est activé, le tableau
      retourné sera vide.
    </ResponseField>

    <ResponseField name="total" type="u64">
      Offre totale en lamports
    </ResponseField>
  </Expandable>
</ResponseField>
