import type { Address, Signature } from "@solana/kit";
import { rpc } from "./config";

export type PaymentCheck = { paid: true } | { paid: false; reason: string };

/**
 * Confirm on your server that `signature` paid this order: the transaction
 * succeeded, it carries the order's reference, and the store's token account
 * for `mint` received exactly `amount` base units. Never trust the browser's
 * word for it.
 */
export async function verifyPayment(order: {
  signature: Signature;
  storeOwner: Address;
  mint: Address;
  amount: bigint;
  reference?: Address;
}): Promise<PaymentCheck> {
  const tx = await rpc
    .getTransaction(order.signature, {
      commitment: "confirmed",
      encoding: "jsonParsed",
      maxSupportedTransactionVersion: 0,
    })
    .send();
  if (!tx) return { paid: false, reason: "transaction not found" };
  if (tx.meta?.err) return { paid: false, reason: "transaction failed" };

  const keys = tx.transaction.message.accountKeys.map((k) => k.pubkey);
  if (order.reference && !keys.includes(order.reference)) {
    return { paid: false, reason: "reference not found" };
  }

  // What the store's token account for this mint held before and after.
  type TokenBalances = NonNullable<typeof tx.meta>["preTokenBalances"];
  const balance = (list: TokenBalances) =>
    BigInt(
      list?.find((b) => b.owner === order.storeOwner && b.mint === order.mint)
        ?.uiTokenAmount.amount ?? "0",
    );
  const received =
    balance(tx.meta?.postTokenBalances) - balance(tx.meta?.preTokenBalances);
  if (received !== order.amount) {
    return {
      paid: false,
      reason: `amount mismatch: expected ${order.amount}, received ${received}`,
    };
  }
  return { paid: true };
}

import asyncio
import os

from solana.rpc.async_api import AsyncClient
from solana.rpc.commitment import Confirmed
from solders.signature import Signature

RPC_URL = os.environ.get("RPC_URL", "https://api.devnet.solana.com")
USDC_MINT = os.environ.get("USDC_MINT", "4zMMC9srt5Ri5X14GAgXhaHii3GnPAEERYPJgZJDncDU")


async def verify_payment(signature: str, store_owner: str, amount: int, reference: str):
    """Return (paid, reason) for a USDC payment of `amount` base units."""
    async with AsyncClient(RPC_URL) as client:
        resp = await client.get_transaction(
            Signature.from_string(signature),
            encoding="jsonParsed",
            commitment=Confirmed,
            max_supported_transaction_version=0,
        )
    tx = resp.value
    if tx is None:
        return False, "transaction not found"
    meta = tx.transaction.meta
    if meta.err is not None:
        return False, "transaction failed"

    keys = [str(k.pubkey) for k in tx.transaction.transaction.message.account_keys]
    if reference not in keys:
        return False, "reference not found"

    def balance(balances):
        for b in balances or []:
            if str(b.owner) == store_owner and str(b.mint) == USDC_MINT:
                return int(b.ui_token_amount.amount)
        return 0

    received = balance(meta.post_token_balances) - balance(meta.pre_token_balances)
    if received != amount:
        return False, f"amount mismatch: expected {amount}, received {received}"
    return True, None
