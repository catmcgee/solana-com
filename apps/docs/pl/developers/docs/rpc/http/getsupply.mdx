Zwraca informacje o obecnej podaży.

<RequestExample>
  ```bash cURL
  curl https://api.devnet.solana.com -s -X POST \
    -H "Content-Type: application/json" \
    -d '
    {
      "jsonrpc": "2.0",
      "id": 1,
      "method": "getSupply",
      "params": [
        {
          "commitment": "finalized"
        }
      ]
    }
  '
  ```

  ```ts Kit
  import { createSolanaRpc } from "@solana/kit";

  const rpc_url = "https://api.devnet.solana.com";
  const rpc = createSolanaRpc(rpc_url);

  let supply = await rpc.getSupply().send();

  console.log(supply);
  ```

  ```ts web3.js
  import { Connection, clusterApiUrl } from "@solana/web3.js";

  const connection = new Connection(clusterApiUrl("devnet"), "confirmed");

  let supply = await connection.getSupply();

  console.log(supply);
  ```
</RequestExample>

## Parameters

<ParamField body="0" type="object">
  Obiekt konfiguracyjny zawierający następujące pola:

  <Expandable title="properties" defaultOpen>
    <ParamField body="commitment" type="string">
      Commitment opisuje, jak bardzo zfinalizowany jest blok w danym momencie. Zobacz
      [Konfigurowanie zaangażowania stanu](/developers/docs/rpc#configuring-state-commitment).
    </ParamField>

    <ParamField body="excludeNonCirculatingAccountsList" type="bool">
      Wyklucz listę kont niebędących w obiegu z odpowiedzi
    </ParamField>
  </Expandable>
</ParamField>

## Result

<ResponseField name="result" type="object">
  Wynik będzie obiektem JSON zawierającym:

  <Expandable title="properties" defaultOpen>
    <ResponseField name="circulating" type="u64">
      Podaż w obiegu w lamportach
    </ResponseField>

    <ResponseField name="nonCirculating" type="u64">
      Podaż nieobiegowa w lamportach
    </ResponseField>

    <ResponseField name="nonCirculatingAccounts" type="array">
      Tablica adresów kont nieobiegowych w formie ciągów znaków. Jeśli
      `excludeNonCirculatingAccountsList` jest włączony, zwrócona tablica będzie
      pusta.
    </ResponseField>

    <ResponseField name="total" type="u64">
      Całkowita podaż w lamportach
    </ResponseField>
  </Expandable>
</ResponseField>
