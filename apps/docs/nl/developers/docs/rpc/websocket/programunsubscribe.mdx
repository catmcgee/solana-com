<RequestExample>
  ```json WebSocket
  {
    "jsonrpc": "2.0",

    "method": "programUnsubscribe",

      0
    ]
  }
  ```
</RequestExample>

<ResponseExample>
  ```json Response

  }
  ```
</ResponseExample>
