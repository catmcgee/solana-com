[주어진 또는 기본 커밋먼트 레벨](/docs/rpc/#configuring-state-commitment)에
도달한 슬롯을 반환합니다

<RequestExample>
  ```bash cURL
  curl https://api.devnet.solana.com -s -X POST \
    -H "Content-Type: application/json" \
    -d '
    {
      "jsonrpc": "2.0",
      "id": 1,
      "method": "getSlot",
      "params": [
        {
          "commitment": "finalized"
        }
      ]
    }
  '
  ```

  ```ts Kit
  import { createSolanaRpc } from "@solana/kit";

  const rpc_url = "https://api.devnet.solana.com";
  const rpc = createSolanaRpc(rpc_url);

  let slot = await rpc.getSlot().send();

  console.log(slot);
  ```

  ```ts web3.js
  import { Connection, clusterApiUrl, type GetSlotConfig } from "@solana/web3.js";

  const connection = new Connection(clusterApiUrl("devnet"), "confirmed");

  let config: GetSlotConfig = {
    commitment: "finalized"
  };

  let slot = await connection.getSlot(config);

  console.log(slot);
  ```

  ```rs Rust
  use anyhow::Result;
  use solana_client::nonblocking::rpc_client::RpcClient;
  use solana_sdk::commitment_config::CommitmentConfig;

  #[tokio::main]
  async fn main() -> Result<()> {
      let client = RpcClient::new_with_commitment(
          String::from("https://api.devnet.solana.com"),
          CommitmentConfig::confirmed(),
      );

      let slot = client.get_slot().await?;

      println!("{}", slot);

      Ok(())
  }
  ```
</RequestExample>
